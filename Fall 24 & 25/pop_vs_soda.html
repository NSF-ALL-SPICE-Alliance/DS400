pop_vs_soda <- pop_vs_soda %>%
  mutate(pop = ifelse(pop, "says pop", "no pop"))

[1] "says pop" "no pop"

Bayes Theorem

\[ P(B \mid A) = \frac{P(B) L(B \mid A)}{P(A)} \]

A <- says pop

  • Ac <- no pop

B <- from the south

  • Bc <- from somewhere else

P(A)
